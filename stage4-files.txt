===== components/shared/MenuCard.tsx =====
"use client";

import { useState } from "react";
import Image from "next/image";
import { Heart, Clock, Minus, Plus } from "lucide-react";
import { motion } from "framer-motion";
import { toast } from "sonner";
import { MenuItem } from "@/types";
import { useCart } from "@/lib/cart-context";
import SoupProteinModal from "@/components/shared/SoupProteinModal";

interface MenuCardProps {
  item: MenuItem;
}

export default function MenuCard({ item }: MenuCardProps) {
  const [qty, setQty] = useState(0);
  const { addItem } = useCart();
  const [showModal, setShowModal] = useState(false);

  const [isFav, setIsFav] = useState(item.isFavorited ?? false);

  const handleAddToCart = () => {
    if (item.requiresProteinSelection) {
      setShowModal(true);
    } else {
      addItem(item);
      toast.success(`${item.name} added to cart`);
    }
  };

  const toggleFavorite = () => {
    setIsFav((prev) => !prev);
    toast(!isFav ? "Added to favorites" : "Removed from favorites", {
      icon: "❤️",
    });
  };

  return (
    <motion.div
      whileHover={{ y: -6 }}
      transition={{ duration: 0.25, ease: [0.25, 0.1, 0.25, 1] }}
      className="relative mt-14 rounded-[20px] bg-white shadow-[0_0_4px_rgba(0,0,0,0.25)]"
    >
      {/* Food / Drink Image */}
      <div className="absolute -top-14 left-1/2 z-10 flex h-[120px] w-[120px] -translate-x-1/2 items-center justify-center">
        <div className="relative h-[120px] w-[120px] overflow-hidden rounded-full bg-gray-light">
          <Image
            src={item.image}
            alt={item.name}
            fill
            className="object-cover"
          />
        </div>
      </div>

      {/* Favorite Icon */}
      <button
        onClick={toggleFavorite}
        className="absolute right-5 top-6 flex h-[50px] w-[50px] items-center justify-center transition-transform hover:scale-110"
        aria-label={isFav ? "Remove from favorites" : "Add to favorites"}
      >
        <Heart
          size={32}
          className={isFav ? "fill-primary-red text-primary-red" : "text-black"}
          strokeWidth={isFav ? 0 : 1}
        />
      </button>

      {/* Card Content */}
      <div className="px-6 pb-6 pt-16">
        {/* Title + Prep Time (food) or Title + Price (drinks) */}
        <div className="flex items-center justify-between">
          <h3 className="font-heading text-xl font-medium">{item.name}</h3>
          {item.prepTime && (
            <div className="flex items-center gap-1 text-foreground/70">
              <Clock size={16} />
              <span className="font-body text-base">{item.prepTime}</span>
            </div>
          )}
          {item.isDrink && (
            <div className="flex items-center gap-1">
              <span className="font-body text-sm text-foreground/50">GH₵</span>
              <span className="font-heading text-xl font-medium">
                {item.price.toFixed(2)}
              </span>
            </div>
          )}
        </div>

        {/* Description (food only) */}
        {item.description && (
          <p className="mt-2 line-clamp-2 font-body text-base text-foreground/70">
            {item.description}
          </p>
        )}

        {/* Bottom row */}
        <div
          className={`flex items-center justify-between ${item.isDrink ? "mt-4" : "mt-6"}`}
        >
          {/* Left side: Price (food) or Quantity controls (drinks) */}
          {item.isDrink ? (
            <div className="flex items-center gap-2">
              <button
                onClick={() => setQty((q) => Math.max(0, q - 1))}
                className="flex h-[25px] w-[25px] items-center justify-center rounded border border-black"
                aria-label="Decrease quantity"
              >
                <Minus size={14} />
              </button>
              <span className="min-w-[16px] text-center font-body text-2xl">
                {qty}
              </span>
              <button
                onClick={() => setQty((q) => q + 1)}
                className="flex h-[25px] w-[25px] items-center justify-center rounded bg-black text-white"
                aria-label="Increase quantity"
              >
                <Plus size={14} />
              </button>
            </div>
          ) : (
            <div className="flex items-center gap-1">
              <span className="font-body text-sm text-foreground/50">GH₵</span>
              <span className="font-heading text-xl font-medium">
                {item.price.toFixed(2)}
              </span>
            </div>
          )}

          <button
            onClick={handleAddToCart}
            className="flex items-center gap-2 rounded-[10px] bg-primary px-5 py-2.5 font-body text-lg text-white transition-colors hover:bg-primary/90"
          >
            {item.requiresProteinSelection ? "select protein" : "add to cart"}
            <span className="text-xl">+</span>
          </button>
        </div>
      </div>

      {/* Soup & Protein selection modal */}
      {showModal && (
        <SoupProteinModal item={item} onClose={() => setShowModal(false)} />
      )}
    </motion.div>
  );
}

// ---------------------------------------------------------------------------
// Types
// ---------------------------------------------------------------------------

export interface ApiErrorResponse {
  message: string;
  statusCode: number;
  errors?: Record<string, string[]>; // field-level validation errors from the API
}

export class ApiError extends Error {
  statusCode: number;
  errors?: Record<string, string[]>;

  constructor(
    message: string,
    statusCode: number,
    errors?: Record<string, string[]>,
  ) {
    super(message);
    this.name = "ApiError";
    this.statusCode = statusCode;
    this.errors = errors;
  }
}

// ---------------------------------------------------------------------------
// Axios instance
// ---------------------------------------------------------------------------

const api: AxiosInstance = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL,
  withCredentials: true, // send/receive httpOnly JWT cookies
  headers: {
    "Content-Type": "application/json",
    "X-API-Key": process.env.NEXT_PUBLIC_API_KEY ?? "",
  },
  timeout: 15_000,
});

// ---------------------------------------------------------------------------
// 401 refresh interceptor
// ---------------------------------------------------------------------------

let isRefreshing = false;
let refreshQueue: Array<{
  resolve: (value: unknown) => void;
  reject: (reason?: unknown) => void;
}> = [];

/** Drain the queue after a refresh attempt */
function drainQueue(error: unknown, token?: string) {
  refreshQueue.forEach(({ resolve, reject }) => {
    if (error) {
      reject(error);
    } else {
      resolve(token);
    }
  });
  refreshQueue = [];
}

api.interceptors.response.use(
  // Pass successful responses straight through
  (response: AxiosResponse) => response,

  async (error: AxiosError<ApiErrorResponse>) => {
    const originalRequest = error.config as InternalAxiosRequestConfig & {
      _retry?: boolean;
    };

    const status = error.response?.status;

    // Only attempt a refresh once per request, and only for 401s
    // Skip refresh if the failing request IS the refresh endpoint (avoid infinite loop)
    if (
      status === 401 &&
      !originalRequest._retry &&
      !originalRequest.url?.includes("/auth/refresh")
    ) {
      if (isRefreshing) {
        // Queue this request until the in-flight refresh resolves
        return new Promise((resolve, reject) => {
          refreshQueue.push({ resolve, reject });
        }).then(() => api(originalRequest));
      }

      originalRequest._retry = true;
      isRefreshing = true;

      try {
        await api.post("/auth/refresh");
        drainQueue(null);
        return api(originalRequest);
      } catch (refreshError) {
        drainQueue(refreshError);
        // Refresh failed — the auth context will handle redirecting to /login
        return Promise.reject(refreshError);
      } finally {
        isRefreshing = false;
      }
    }

    // Normalize all API errors into ApiError instances
    const message =
      error.response?.data?.message ??
      error.message ??
      "An unexpected error occurred";
    const statusCode = error.response?.status ?? 0;
    const errors = error.response?.data?.errors;

    return Promise.reject(new ApiError(message, statusCode, errors));
  },
);

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
} from "react";
import { useRouter } from "next/navigation";
import { get, post, ApiError } from "@/lib/api";

// ---------------------------------------------------------------------------
// Types — matching GET /auth/me response shape exactly
// ---------------------------------------------------------------------------

export interface UserRole {
  _id: string;
  name: string;
  permissions: string[];
}

export interface User {
  _id: string;
  name: string;
  email: string;
  authMethod: "local" | "google" | "apple";
  role: UserRole;
  phoneNumber?: string;
  avatar: string | null;
  emailVerified: boolean;
  addresses: string[];
  defaultAddress?: string;
  active: boolean;
  lastLoginAt: string;
  createdAt: string;
}

// ---------------------------------------------------------------------------
// Request / Response payload types
// ---------------------------------------------------------------------------

interface SignupPayload {
  name: string;
  email: string;
  password: string;
  passwordConfirm: string;
  phoneNumber?: string;
}

interface LoginPayload {
  email: string;
  password: string;
}

interface UpdateProfilePayload {
  name?: string;
  phoneNumber?: string;
  avatar?: string;
}

interface AuthMeResponse {
  status: string;
  data: { user: User };
}

interface SignupResponse {
  status: string;
  message: string;
  data: { user: User };
}

// ---------------------------------------------------------------------------
// Context shape
// ---------------------------------------------------------------------------

interface AuthContextValue {
  user: User | null;
  isLoading: boolean;          // true while the initial /auth/me check is in flight
  isAuthenticated: boolean;
  isAdmin: boolean;            // true if role.name !== "customer"
  hasPermission: (permission: string) => boolean;
  login: (payload: LoginPayload) => Promise<void>;
  signup: (payload: SignupPayload) => Promise<User>;
  logout: () => Promise<void>;
  googleLogin: (idToken: string) => Promise<void>;
  appleLogin: (identityToken: string, user?: AppleUser) => Promise<void>;
  refreshUser: () => Promise<void>;  // re-fetches /auth/me — useful after profile updates
}

interface AppleUser {
  name?: { firstName?: string; lastName?: string };
  email?: string;
}

// ---------------------------------------------------------------------------
// Context
// ---------------------------------------------------------------------------

const AuthContext = createContext<AuthContextValue | null>(null);

// ---------------------------------------------------------------------------
// Provider
// ---------------------------------------------------------------------------

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const router = useRouter();

  // Fetch the current user — called on mount and after profile updates
  const fetchUser = useCallback(async () => {
    try {
      const response = await get<AuthMeResponse>("/auth/me");
      setUser(response.data.user);
    } catch (error) {
      // 401 means no valid session — that's expected for logged-out users
      if (error instanceof ApiError && error.statusCode === 401) {
        setUser(null);
      } else {
        // Unexpected error — still clear the user so the UI stays consistent
        setUser(null);
        console.error("[AuthContext] Failed to fetch user:", error);
      }
    } finally {
      setIsLoading(false);
    }
  }, []);

  // On mount: check if a session already exists (cookie may still be valid)
  useEffect(() => {
    fetchUser();
  }, [fetchUser]);

  // ── Auth actions ──────────────────────────────────────────────────────────

  const login = useCallback(
    async (payload: LoginPayload) => {
      // POST /auth/login sets cookies automatically — no token in response body
      await post("/auth/login", payload);
      // Fetch the user profile now that cookies are set
      await fetchUser();
    },
    [fetchUser]
  );

  const signup = useCallback(async (payload: SignupPayload): Promise<User> => {
    const response = await post<SignupResponse>("/auth/signup", payload);
    // Signup also sets cookies — user is logged in immediately
    setUser(response.data.user);
    return response.data.user;
  }, []);

  const logout = useCallback(async () => {
    try {
      await post("/auth/logout");
    } finally {
      // Always clear local state even if the request fails
      setUser(null);
      router.push("/");
    }
  }, [router]);

  const googleLogin = useCallback(
    async (idToken: string) => {
      await post("/auth/google", { idToken });
      await fetchUser();
    },
    [fetchUser]
  );

  const appleLogin = useCallback(
    async (identityToken: string, appleUser?: AppleUser) => {
      await post("/auth/apple", { identityToken, user: appleUser });
      await fetchUser();
    },
    [fetchUser]
  );

  const refreshUser = useCallback(async () => {
    await fetchUser();
  }, [fetchUser]);

  // ── Derived state ─────────────────────────────────────────────────────────

  const isAuthenticated = user !== null;

  // Any role other than "customer" (e.g. "admin", "super_admin", "rider") is
  // considered an admin for the purpose of showing the admin panel
  const isAdmin = isAuthenticated && user!.role.name !== "customer";

  const hasPermission = useCallback(
    (permission: string) => {
      if (!user) return false;
      return user.role.permissions.includes(permission);
    },
    [user]
  );

  // ── Context value (memoized to prevent unnecessary re-renders) ────────────

  const value = useMemo<AuthContextValue>(
    () => ({
      user,
      isLoading,
      isAuthenticated,
      isAdmin,
      hasPermission,
      login,
      signup,
      logout,
      googleLogin,
      appleLogin,
      refreshUser,
    }),
    [
      user,
      isLoading,
      isAuthenticated,
      isAdmin,
      hasPermission,
      login,
      signup,
      logout,
      googleLogin,
      appleLogin,
      refreshUser,
    ]
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

// ---------------------------------------------------------------------------
// Hook
// ---------------------------------------------------------------------------

export function useAuth(): AuthContextValue {
  const context = useContext(AuthContext);
  if (!context) {
    throw new Error("useAuth must be used inside <AuthProvider>");
  }
  return context;
}

// 📁 lib/cart-context.tsx

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
} from "react";
import { get, post, patch, del } from "@/lib/api";
import { useAuth } from "@/lib/auth-context";
import type { Cart, CartItem, LocalCartItem, MenuItem } from "@/types";

// ---------------------------------------------------------------------------
// Types
// ---------------------------------------------------------------------------

interface CartContextValue {
  // Items — unified view regardless of auth state
  items: LocalCartItem[];
  totalAmount: number;
  itemCount: number;
  isLoading: boolean;

  // Actions
  addItem: (
    item: MenuItem,
    quantity?: number,
    customization?: string,
  ) => Promise<void>;
  updateQuantity: (menuItemId: string, quantity: number) => Promise<void>;
  removeItem: (menuItemId: string) => Promise<void>;
  clearCart: () => Promise<void>;

  // Cart panel UI state
  isOpen: boolean;
  openCart: () => void;
  closeCart: () => void;
}

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

const LOCAL_CART_KEY = "ek_cart";

function loadLocalCart(): LocalCartItem[] {
  if (typeof window === "undefined") return [];
  try {
    const raw = localStorage.getItem(LOCAL_CART_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

function saveLocalCart(items: LocalCartItem[]) {
  if (typeof window === "undefined") return;
  localStorage.setItem(LOCAL_CART_KEY, JSON.stringify(items));
}

function serverCartToLocal(cart: Cart): LocalCartItem[] {
  return cart.items.map((item: CartItem) => ({
    menuItemId: item.menuItem._id,
    name: item.menuItem.name,
    price: item.unitPrice,
    image: item.menuItem.images?.[0] ?? "",
    quantity: item.quantity,
  }));
}

// ---------------------------------------------------------------------------
// Context
// ---------------------------------------------------------------------------

const CartContext = createContext<CartContextValue | null>(null);

// ---------------------------------------------------------------------------
// Provider
// ---------------------------------------------------------------------------

export function CartProvider({ children }: { children: React.ReactNode }) {
  const { isAuthenticated } = useAuth();
  const [items, setItems] = useState<LocalCartItem[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [isOpen, setIsOpen] = useState(false);

  // Track previous auth state to detect login events
  const prevAuthRef = useRef(isAuthenticated);

  // ── Load cart ─────────────────────────────────────────────────────────────

  const loadServerCart = useCallback(async () => {
    setIsLoading(true);
    try {
      const res = await get<{ status: string; data: { cart: Cart } }>("/cart");
      setItems(serverCartToLocal(res.data.cart));
    } catch {
      // Empty or error — keep current items
    } finally {
      setIsLoading(false);
    }
  }, []);

  // On mount: load from localStorage if not authenticated
  useEffect(() => {
    if (!isAuthenticated) {
      setItems(loadLocalCart());
    }
  }, []); // eslint-disable-line react-hooks/exhaustive-deps

  // On login: merge local cart into server cart, then load server cart
  useEffect(() => {
    const wasLoggedOut = !prevAuthRef.current;
    const isNowLoggedIn = isAuthenticated;

    if (wasLoggedOut && isNowLoggedIn) {
      const localItems = loadLocalCart();

      const mergeAndLoad = async () => {
        setIsLoading(true);
        try {
          // Add each local item to the server cart
          for (const item of localItems) {
            await post("/cart/items", {
              menuItem: item.menuItemId,
              quantity: item.quantity,
            }).catch(() => {}); // best-effort — skip items that fail
          }
          // Clear local storage after merge
          saveLocalCart([]);
          // Load the merged server cart
          await loadServerCart();
        } finally {
          setIsLoading(false);
        }
      };

      mergeAndLoad();
    }

    // On logout: switch to local cart (will be empty since we cleared it)
    if (!isAuthenticated && prevAuthRef.current) {
      setItems(loadLocalCart());
    }

    prevAuthRef.current = isAuthenticated;
  }, [isAuthenticated, loadServerCart]);

  // Persist local cart to localStorage whenever items change (anonymous only)
  useEffect(() => {
    if (!isAuthenticated) {
      saveLocalCart(items);
    }
  }, [items, isAuthenticated]);

  // ── Cart actions ──────────────────────────────────────────────────────────

  const addItem = useCallback(
    async (menuItem: MenuItem, quantity = 1, customization?: string) => {
      if (isAuthenticated) {
        setIsLoading(true);
        try {
          const res = await post<{ status: string; data: { cart: Cart } }>(
            "/cart/items",
            { menuItem: menuItem._id, quantity },
          );
          setItems(serverCartToLocal(res.data.cart));
        } finally {
          setIsLoading(false);
        }
      } else {
        setItems((prev) => {
          const existing = prev.find((i) => i.menuItemId === menuItem._id);
          if (existing) {
            return prev.map((i) =>
              i.menuItemId === menuItem._id
                ? { ...i, quantity: i.quantity + quantity }
                : i,
            );
          }
          return [
            ...prev,
            {
              menuItemId: menuItem._id,
              name: menuItem.name,
              price: menuItem.price,
              image: menuItem.images?.[0] ?? "",
              quantity,
              customization,
            },
          ];
        });
      }
    },
    [isAuthenticated],
  );

  const updateQuantity = useCallback(
    async (menuItemId: string, quantity: number) => {
      if (quantity < 1) return;

      if (isAuthenticated) {
        setIsLoading(true);
        try {
          const res = await patch<{ status: string; data: { cart: Cart } }>(
            `/cart/items/${menuItemId}`,
            { quantity },
          );
          setItems(serverCartToLocal(res.data.cart));
        } finally {
          setIsLoading(false);
        }
      } else {
        setItems((prev) =>
          prev.map((i) =>
            i.menuItemId === menuItemId ? { ...i, quantity } : i,
          ),
        );
      }
    },
    [isAuthenticated],
  );

  const removeItem = useCallback(
    async (menuItemId: string) => {
      if (isAuthenticated) {
        setIsLoading(true);
        try {
          const res = await del<{ status: string; data: { cart: Cart } }>(
            `/cart/items/${menuItemId}`,
          );
          setItems(serverCartToLocal(res.data.cart));
        } finally {
          setIsLoading(false);
        }
      } else {
        setItems((prev) => prev.filter((i) => i.menuItemId !== menuItemId));
      }
    },
    [isAuthenticated],
  );

  const clearCart = useCallback(async () => {
    if (isAuthenticated) {
      setIsLoading(true);
      try {
        await del("/cart");
        setItems([]);
      } finally {
        setIsLoading(false);
      }
    } else {
      setItems([]);
      saveLocalCart([]);
    }
  }, [isAuthenticated]);

  // ── Derived values ────────────────────────────────────────────────────────

  const totalAmount = useMemo(
    () => items.reduce((sum, item) => sum + item.price * item.quantity, 0),
    [items],
  );

  const itemCount = useMemo(
    () => items.reduce((sum, item) => sum + item.quantity, 0),
    [items],
  );

  const openCart = useCallback(() => setIsOpen(true), []);
  const closeCart = useCallback(() => setIsOpen(false), []);

  const value = useMemo<CartContextValue>(
    () => ({
      items,
      totalAmount,
      itemCount,
      isLoading,
      addItem,
      updateQuantity,
      removeItem,
      clearCart,
      isOpen,
      openCart,
      closeCart,
    }),
    [
      items,
      totalAmount,
      itemCount,
      isLoading,
      addItem,
      updateQuantity,
      removeItem,
      clearCart,
      isOpen,
      openCart,
      closeCart,
    ],
  );

  return <CartContext.Provider value={value}>{children}</CartContext.Provider>;
}

  if (!context) {

  }
  return context;
}

    enabled: Boolean(id),
    // Poll every 30s for active orders so status updates appear automatically
    refetchInterval: (query) => {
      const order = query.state.data;
      if (!order) return false;
      const activeStatuses = [
        "pending",
        "confirmed",
        "preparing",
        "ready_for_pickup",
        "out_for_delivery",
      ];
      return activeStatuses.includes(order.status) ? 30_000 : false;
    },
  });
}

// ---------------------------------------------------------------------------
// Place order
// ---------------------------------------------------------------------------

export function usePlaceOrder() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (payload: CreateOrderPayload) =>
      post<ApiResponse<{ order: Order }>>("/orders", payload).then(
        (res) => res.data.order,
      ),
    onSuccess: () => {
      // Invalidate order history so the new order appears
      queryClient.invalidateQueries({ queryKey: orderKeys.myOrders() });
      // Cart is cleared server-side after order — invalidate cart cache too
      queryClient.invalidateQueries({ queryKey: ["cart"] });
    },
  });
}

// ---------------------------------------------------------------------------
// Cancel order
// ---------------------------------------------------------------------------

export function useCancelOrder() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (id: string) => post(`/orders/${id}/cancel`),
    onSuccess: (_data, id) => {
      queryClient.invalidateQueries({ queryKey: orderKeys.detail(id) });
      queryClient.invalidateQueries({ queryKey: orderKeys.myOrders() });
    },
  });
}
